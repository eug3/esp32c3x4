int display_draw_text_menu(int x, int y, const char *text, uint8_t color, uint8_t bg_color)
{
    if (text == NULL || *text == '\0') {
        return 0;
    }

    ESP_LOGI(TAG, "display_draw_text_menu: drawing '%s' at (%d,%d)", text, x, y);

    sFONT *font = display_get_menu_font();
    if (font == NULL) {
        ESP_LOGE(TAG, "Menu font is NULL! Using fallback Font16");
        font = &Font16;
    }
    
    ESP_LOGI(TAG, "Font: %p, Width=%d, Height=%d, table=%p", 
             (void*)font, font->Width, font->Height, (void*)font->table);
    
    // 检查字体表是否有效
    if (font->table == NULL) {
        ESP_LOGE(TAG, "Font table is NULL! Cannot draw text");
        return 0;
    }
    
    // 检查 framebuffer
    if (s_framebuffer == NULL) {
        ESP_LOGE(TAG, "Framebuffer is NULL! Cannot draw text");
        return 0;
    }
    
    // 检查 Paint.Image
    if (Paint.Image == NULL) {
        ESP_LOGE(TAG, "Paint.Image is NULL! Attempting to fix...");
        Paint_SelectImage(s_framebuffer);
        if (Paint.Image == NULL) {
            ESP_LOGE(TAG, "Cannot fix Paint.Image! Aborting draw");
            return 0;
        }
    }

    lock_engine();
    int width;
    if (text_has_non_ascii(text)) {
        width = draw_text_utf8_menu_locked(x, y, text, font, color, bg_color);
    } else {
        const int ascii_adv = (int)font->Width;
        int cur_x = x;
        for (const char *p = text; *p != '\0'; p++) {
            Paint_DrawChar(cur_x, y, *p, font, color, bg_color);
            cur_x += ascii_adv;
        }
        width = cur_x - x;

        if (s_config.use_partial_refresh) {
            expand_dirty_region(x, y, width, font->Height);
        }
    }
    unlock_engine();
    
    ESP_LOGI(TAG, "display_draw_text_menu: completed, width=%d", width);

    return width;
}
